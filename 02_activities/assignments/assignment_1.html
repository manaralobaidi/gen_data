# Plot
p1 <- ggplot() +
  geom_point(
    data = count_data,
    aes(x = rs1861, y = PHENOTYPE, size = count),
    color = "blue"
  ) +
  geom_line(
    data = geno_range,
    aes(x = rs1861, y = predicted),
    color = "red",
    size = 1.2
  ) +
  scale_size_continuous(range = c(2, 10)) +
  labs(
    title = "Linear Regression: PHENOTYPE vs rs1861",
    x = "Genotype (0/1/2)",
    y = "Phenotype",
    size = "Count"
  ) +
  coord_cartesian(xlim = c(-0.5, 2.5)) +
  theme_minimal()
